if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT TARGET matterm::pty)
    message(FATAL_ERROR "matterm-server requires Linux and MATTERM_ENABLE_PTY=ON")
endif()
find_package(Boost 1.74 REQUIRED)
find_package(Threads REQUIRED)
find_path(SODIUM_INCLUDE_DIR sodium.h REQUIRED)
if(MATTERM_STATIC_DAEMON)
    # Its own cache entry: switching the option must not reuse the other kind of library.
    find_library(SODIUM_LIBRARY_STATIC NAMES libsodium.a REQUIRED)
    set(SODIUM_LIBRARY ${SODIUM_LIBRARY_STATIC})
else()
    find_library(SODIUM_LIBRARY_SHARED sodium REQUIRED)
    set(SODIUM_LIBRARY ${SODIUM_LIBRARY_SHARED})
endif()
if(NOT EXISTS "${PROJECT_SOURCE_DIR}/lib/fast-json/include/fast-json/json.h")
    message(FATAL_ERROR "matterm-server reads its configuration with lib/fast-json: run 'git submodule update --init lib/fast-json'")
endif()
# The daemon's wire format (framing and control messages), shared by the daemon and by the
# server, which is a client of the daemon with --expose-local-sessions.
add_library(matterm-daemon-wire ${PROJECT_SOURCE_DIR}/src/daemon/control.cpp ${PROJECT_SOURCE_DIR}/src/daemon/socketpath.cpp ${PROJECT_SOURCE_DIR}/src/daemon/buildhash.cpp)
target_compile_features(matterm-daemon-wire PUBLIC cxx_std_17)
target_include_directories(matterm-daemon-wire PUBLIC ${PROJECT_SOURCE_DIR}/src/daemon)
target_link_libraries(matterm-daemon-wire PUBLIC matterm::protocol)
add_library(matterm-server-core auth.cpp config.cpp ratelimiter.cpp proxysource.cpp replayring.cpp sessionregistry.cpp ptysessionhost.cpp server.cpp localsessions.cpp)
# fast-json defines non-inline specializations: include it from config.cpp only.
target_include_directories(matterm-server-core PRIVATE ${PROJECT_SOURCE_DIR}/lib/fast-json/include)
target_compile_features(matterm-server-core PUBLIC cxx_std_20)
target_include_directories(matterm-server-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${Boost_INCLUDE_DIRS} ${SODIUM_INCLUDE_DIR})
target_compile_definitions(matterm-server-core PUBLIC BOOST_ERROR_CODE_HEADER_ONLY)
target_link_libraries(matterm-server-core PUBLIC matterm-daemon-wire matterm::protocol matterm::pty Threads::Threads ${SODIUM_LIBRARY})
add_executable(matterm-server main.cpp)
target_link_libraries(matterm-server PRIVATE matterm-server-core)
